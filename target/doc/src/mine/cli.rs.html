10/// Simple CLI frontend for the Minesweeper board logic.
11///
12/// This function starts a minimal interactive REPL that accepts commands to reveal and flag cells. It's intentionally small and avoids any external dependencies so it can be used as a starting point for both tests and integration with other frontends.
13///
14/// Possible commands:
15/// - `reveal x y`: Reveal the cell at position (x, y).
16/// - `flag x y`: Flag the cell at position (x, y).
17/// - `help`: Display a list of available commands.
18/// - `quit`: Quit the game.
19pub fn run(rows: usize, cols: usize, bomb_coords: Vec<(usize, usize)>) {
20    let mut board = Board::from_bomb_coords((rows, cols), &bomb_coords);
21
22    println!("Minesweeper CLI");
23    println!("Board size: {} rows x {} cols", rows, cols);
24    println!("Commands:");
25    println!("  r <row> <col>  - reveal at (row,col)");
26    println!("  f <row> <col>  - cycle flag at (row,col) (Hidden -> Flag -> Question -> Hidden)");
27    println!("  h              - help");
28    println!("  q              - quit");
29    println!("Rows and cols are 0-based indices (0..rows-1, 0..cols-1).");
30    println!();
31
32    loop {
33        print_board_console(&board);
34        if board.game_over {
35            println!("Game over. A bomb was revealed.");
36            break;
37        }
38        if board.check_win() {
39            println!("You won! All safe cells revealed.");
40            break;
41        }
42
43        print!("> ");
44        let _ = io::stdout().flush();
45
46        let mut input = String::new();
47        if io::stdin().read_line(&mut input).is_err() {
48            println!("Failed to read input - exiting.");
49            break;
50        }
51        let input = input.trim();
52        if input.is_empty() {
53            continue;
54        }
55
56        let parts: Vec<&str> = input.split_whitespace().collect();
57        match parts[0].to_lowercase().as_str() {
58            "q" | "quit" | "exit" => {
59                println!("Bye.");
60                break;
61            }
62            "h" | "help" => {
63                println!("Commands:");
64                println!("  r <row> <col>  - reveal at (row,col)");
65                println!(
66                    "  f <row> <col>  - cycle flag at (row,col) (Hidden -> Flag -> Question -> Hidden)"
67                );
68                println!("  h              - help");
69                println!("  q              - quit");
70                println!("Rows and cols are 0-based indices (0..rows-1, 0..cols-1).");
71                println!();
72            }
73            "r" | "reveal" => {
74                if parts.len() < 3 {
75                    println!("Usage: r <row> <col>");
76                    continue;
77                }
78                let rr = match parts[1].parse::<usize>() {
79                    Ok(v) => v,
80                    Err(_) => {
81                        println!("Invalid row: {}", parts[1]);
82                        continue;
83                    }
84                };
85                let cc = match parts[2].parse::<usize>() {
86                    Ok(v) => v,
87                    Err(_) => {
88                        println!("Invalid col: {}", parts[2]);
89                        continue;
90                    }
91                };
92                match board.reveal_rc(rr, cc) {
93                    Ok(n) => {
94                        println!("Revealed ({},{}) => {}", rr, cc, n);
95                    }
96                    Err(e) if e == "FLAGGED" => {
97                        println!("Cell ({},{}) is flagged. Unflag before revealing.", rr, cc);
98                    }
99                    Err(e) if e == "OUT_OF_BOUNDS" => {
100                        println!("Coordinates out of bounds.");
101                    }
102                    Err(e) if e == "BOMB" => {
103                        println!("Boom! You revealed a bomb at ({},{})", rr, cc);
104                    }
105                    Err(e) => {
106                        println!("Reveal failed: {}", e);
107                    }
108                }
109            }
110            "f" | "flag" | "mark" => {
111                if parts.len() < 3 {
112                    println!("Usage: f <row> <col>");
113                    continue;
114                }
115                let rr = match parts[1].parse::<usize>() {
116                    Ok(v) => v,
117                    Err(_) => {
118                        println!("Invalid row: {}", parts[1]);
119                        continue;
120                    }
121                };
122                let cc = match parts[2].parse::<usize>() {
123                    Ok(v) => v,
124                    Err(_) => {
125                        println!("Invalid col: {}", parts[2]);
126                        continue;
127                    }
128                };
129                match board.cycle_flag_rc(rr, cc) {
130                    Ok(()) => {
131                        println!("Cycled mark at ({},{})", rr, cc);
132                    }
133                    Err(e) if e == "OUT_OF_BOUNDS" => {
134                        println!("Coordinates out of bounds.");
135                    }
136                    Err(e) if e == "REVEALED" => {
137                        println!("Cell is already revealed.");
138                    }
139                    Err(e) => {
140                        println!("Failed to change mark: {}", e);
141                    }
142                }
143            }
144            other => {
145                println!("Unknown command: {}. Use r/f/q.", other);
146            }
147        }
148    }
149}

151/// Render the board to a multi-line string and print it.
152///
153/// It has the column number and row number at the top and sides for the players to be able to identify the position of the cells withe ease
154///
155/// This representation is in the cli, and therefore consists of a simple ASCII view:
156///  - Hidden cells: '.'
157///  - Flagged cells: 'F'
158///  - Question: '?'
159///  - Revealed bomb: 'B'
160///  - Revealed number: ' ' for 0, digit for 1..8 according to the number of surrounding bombs
161fn print_board_console(board: &Board) {
162    let (rows, cols) = board.size;
163    let row_num_chars = (rows - 1).to_string().len();
164    let col_num_chars = (cols - 1).to_string().len();
165    println!();
166    println!(
167        "{:>row_num_chars$} |{}",
168        "",
169        (0..cols)
170            .into_iter()
171            .map(|x| format!(" {:0>col_num_chars$}", x))
172            .collect::<String>()
173    );
174    println!(
175        "{:->row_num_chars$}--{}",
176        "-",
177        (0..cols)
178            .into_iter()
179            .map(|_| format!("-{:-<col_num_chars$}", '-'))
180            .collect::<String>()
181    );
182    for r in 0..rows {
183        let mut line = String::with_capacity(cols * 2);
184        for c in 0..cols {
185            let idx = r * cols + c;
186            let cell = &board.board[idx];
187            let ch = match cell.status {
188                Status::Hidden => '.',
189                Status::Flag => 'F',
190                Status::Question => '?',
191                Status::Revealed => match cell.cell_type {
192                    CellType::Bomb => 'B',
193                    CellType::Number(0) => ' ',
194                    CellType::Number(n) if (1..=9).contains(&n) => {
195                        std::char::from_digit(n as u32, 10).unwrap_or('#')
196                    }
197                    CellType::Number(_) => '#',
198                },
199            };
200            if c < cols {
201                line.push_str(format!("{:<col_num_chars$}", " ").as_str());
202            }
203            line.push(ch);
204        }
205        println!("{:0>row_num_chars$} |{}", r, line);
206    }
207    println!();
208}
