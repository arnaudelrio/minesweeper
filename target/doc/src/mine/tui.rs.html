1//! TUI implementation of the game.
2//!
3//! This module provides a full TUI implementation using the `crossterm` and `tui` crates.
4//!
5//! It exposes a function `run_tui` that starts the TUI game.
6
7#[cfg(feature = "mine_tui")]
8mod enabled {
9    use std::error::Error;
10    use std::io;
11    use std::time::Duration;
12
13    use crossterm::{
14        event::{self, Event as CEvent, KeyCode, KeyEvent},
15        execute,
16        terminal::{EnterAlternateScreen, LeaveAlternateScreen, disable_raw_mode, enable_raw_mode},
17    };
18    use tui::{
19        Terminal,
20        backend::CrosstermBackend,
21        layout::{Alignment, Constraint, Direction, Layout, Rect},
22        style::{Color, Modifier, Style},
23        text::{Span, Spans},
24        widgets::{Block, Borders, Clear, Paragraph},
25    };
26
27    use crate::cells::{CellType, Status};
28    use crate::game_play::Board;
29
30    /// Full TUI implementation.
31    pub fn run_tui(
32        rows: usize,
33        cols: usize,
34        bomb_coords: Vec<(usize, usize)>,
35    ) -> Result<(), Box<dyn Error>> {
36        enable_raw_mode()?;
37        let mut stdout = io::stdout();
38        execute!(stdout, EnterAlternateScreen)?;
39        let backend = CrosstermBackend::new(stdout);
40        let mut terminal = Terminal::new(backend)?;
41
42        let num_bombs = bomb_coords.len();
43        let mut board = Board::from_bomb_coords((rows, cols), &bomb_coords);
44
45        let mut cursor_r: usize = 0;
46        let mut cursor_c: usize = 0;
47
48        loop {
49            let game_over = board.game_over;
50            let win = board.check_win();
51            let game_ended = game_over || win;
52
53            terminal.draw(|f| {
54                let size = f.size();
55                let chunks = Layout::default()
56                    .direction(Direction::Vertical)
57                    .constraints([Constraint::Min(3), Constraint::Length(3)].as_ref())
58                    .split(size);
59
60                let (rows, cols) = board.size;
61                let mut lines = Vec::with_capacity(rows);
62                for r in 0..rows {
63                    let mut spans = Vec::with_capacity(cols * 2);
64                    for c in 0..cols {
65                        let idx = r * cols + c;
66                        let cell = &board.board[idx];
67
68                        let (ch, color) = match cell.status {
69                            Status::Hidden => ('.', Color::White),
70                            Status::Flag => ('F', Color::Red),
71                            Status::Question => ('?', Color::Yellow),
72                            Status::Revealed => match cell.cell_type {
73                                CellType::Bomb => ('B', Color::Red),
74                                CellType::Number(0) => (' ', Color::White),
75                                CellType::Number(n) if (1..=9).contains(&n) => {
76                                    (std::char::from_digit(n as u32, 10).unwrap_or('#'), Color::White)
77                                }
78                                CellType::Number(_) => ('#', Color::Red),
79                            },
80                        };
81
82                        if r == cursor_r && c == cursor_c {
83                            spans.push(Span::styled(
84                                format!("{} ", ch),
85                                Style::default().fg(color).bg(Color::Yellow).add_modifier(Modifier::BOLD),
86                            ));
87                        } else {
88                            spans.push(Span::styled(
89                                format!("{} ", ch),
90                                Style::default().fg(color),
91                            ));
92                        }
93                    }
94                    lines.push(Spans::from(spans));
95                }
96
97                let board_block = Paragraph::new(lines).block(Block::default().borders(Borders::ALL).title("Mines (TUI)"));
98                f.render_widget(board_block, chunks[0]);
99
100                let info_text = if game_ended {
101                    "Press r to restart • q to quit"
102                } else {
103                    "Controls: arrow keys/hjkl to move • Space/Enter to reveal • f to flag • r to restart • q to quit"
104                };
105
106                let info_lines = vec![
107                    Spans::from(vec![Span::raw(info_text)]),
108                    Spans::from(vec![Span::raw(format!(
109                        "Cursor: ({},{})  •  Game over: {}  •  Win: {}",
110                        cursor_r,
111                        cursor_c,
112                        board.game_over,
113                        board.check_win()
114                    ))]),
115                ];
116                let info_block = Paragraph::new(info_lines).block(Block::default().borders(Borders::ALL).title("Info"));
117                f.render_widget(info_block, chunks[1]);
118
119                if game_ended {
120                    let area = centered_rect(60, 20, size);
121                    f.render_widget(Clear, area);
122
123                    let (title, color, msg) = if win {
124                        ("YOU WON!", Color::Green, "Congratulations! You cleared the board.")
125                    } else {
126                        ("GAME OVER", Color::Red, "Boom! You hit a bomb.")
127                    };
128
129                    let block = Block::default()
130                        .title(title)
131                        .borders(Borders::ALL)
132                        .style(Style::default().fg(color));
133
134                    let text = vec![
135                        Spans::from(Span::styled(msg, Style::default().add_modifier(Modifier::BOLD))),
136                        Spans::from(""),
137                        Spans::from("Press 'r' to restart"),
138                        Spans::from("Press 'q' to quit"),
139                    ];
140
141                    let paragraph = Paragraph::new(text)
142                        .block(block)
143                        .alignment(Alignment::Center);
144
145                    f.render_widget(paragraph, area);
146                }
147            })?;
148
149            if event::poll(Duration::from_millis(200))? {
150                match event::read()? {
151                    CEvent::Key(KeyEvent { code, .. }) => match code {
152                        KeyCode::Char('q') | KeyCode::Char('Q') => {
153                            break;
154                        }
155                        KeyCode::Char('r') | KeyCode::Char('R') => {
156                            let new_bombs = Board::random_bomb_coords((rows, cols), num_bombs);
157                            board = Board::from_bomb_coords((rows, cols), &new_bombs);
158                            cursor_r = 0;
159                            cursor_c = 0;
160                        }
161                        _ if !game_ended => match code {
162                            KeyCode::Left | KeyCode::Char('h') => {
163                                if cursor_c > 0 {
164                                    cursor_c -= 1;
165                                }
166                            }
167                            KeyCode::Right | KeyCode::Char('l') => {
168                                if cursor_c + 1 < board.size.1 {
169                                    cursor_c += 1;
170                                }
171                            }
172                            KeyCode::Up | KeyCode::Char('k') => {
173                                if cursor_r > 0 {
174                                    cursor_r -= 1;
175                                }
176                            }
177                            KeyCode::Down | KeyCode::Char('j') => {
178                                if cursor_r + 1 < board.size.0 {
179                                    cursor_r += 1;
180                                }
181                            }
182                            KeyCode::Char('f') | KeyCode::Char('F') => {
183                                let _ = board.cycle_flag_rc(cursor_r, cursor_c);
184                            }
185                            KeyCode::Enter | KeyCode::Char(' ') => {
186                                let _ = board.reveal_rc(cursor_r, cursor_c);
187                            }
188                            _ => { /* ignore other keys */ }
189                        },
190                        _ => { /* ignore keys when game ended */ }
191                    },
192                    _ => { /* ignore mouse/resize events in this stub */ }
193                }
194            }
195        }
196
197        disable_raw_mode()?;
198        execute!(terminal.backend_mut(), LeaveAlternateScreen)?;
199        terminal.show_cursor()?;
200
201        Ok(())
202    }
203
204    /// Helper function to center a rect to show the game over message
205    fn centered_rect(percent_x: u16, percent_y: u16, r: Rect) -> Rect {
206        let popup_layout = Layout::default()
207            .direction(Direction::Vertical)
208            .constraints(
209                [
210                    Constraint::Percentage((100 - percent_y) / 2),
211                    Constraint::Percentage(percent_y),
212                    Constraint::Percentage((100 - percent_y) / 2),
213                ]
214                .as_ref(),
215            )
216            .split(r);
217
218        Layout::default()
219            .direction(Direction::Horizontal)
220            .constraints(
221                [
222                    Constraint::Percentage((100 - percent_x) / 2),
223                    Constraint::Percentage(percent_x),
224                    Constraint::Percentage((100 - percent_x) / 2),
225                ]
226                .as_ref(),
227            )
228            .split(popup_layout[1])[1]
229    }

232/// Terminal User Interface (TUI) for the Minesweeper game, when feature is enabled
233#[cfg(feature = "mine_tui")]
234pub use enabled::run_tui;
235
236/// Function to prevent errors when TUI is not enabled
237#[cfg(not(feature = "mine_tui"))]
238pub fn run_tui(
239    _: usize,
240    _: usize,
241    _: Vec<(usize, usize)>,
242) -> Result<(), Box<dyn std::error::Error>> {
243    println!(
244        "TUI is not enabled.\nPlease enable the feature by running `cargo run --features mine_tui -- tui <args>`."
245    );
246    Ok(())
