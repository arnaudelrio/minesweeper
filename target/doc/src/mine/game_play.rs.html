9/// Game Board structure
10///
11/// Every new game uses an instance of this struct.
12///
13/// This is the main structure used in the game and it basically consists of an array of Cells.
14pub struct Board {
15    /// Board size: rows, cols
16    pub size: (usize, usize),
17    /// Board cells. It should be of length `size.0` * `size.1`
18    pub board: Vec<Cell>,
19    /// Game Over status
20    pub game_over: bool,
21}
22
23/// This is the implementation of the Board struct.
24///
25/// It provides methods for creating a new game board and generating random bomb coordinates, as well as revealing cells.
26impl Board {
27    /// Create an empty board (all cells are Number(0) and Hidden).
28    ///
29    /// Use [`Board::from_bomb_coords()`] to create a board that contains bombs.
30    pub fn new_game(size: (usize, usize)) -> Board {
31        let (rows, cols) = size;
32        let mut board = Vec::with_capacity(rows * cols);
33        for _ in 0..rows * cols {
34            board.push(Cell::new_number(0));
35        }
36        Board {
37            size,
38            board,
39            game_over: false,
40        }
41    }
42
43    /// Generate random coordinates for bombs.
44    ///
45    /// The function uses a random number generator to select unique coordinates within the given range.
46    pub fn random_bomb_coords(size: (usize, usize), num_bombs: usize) -> Vec<(usize, usize)> {
47        let mut rng = rand::rng();
48        let mut bomb_coords = Vec::with_capacity(num_bombs);
49        while bomb_coords.len() < num_bombs {
50            let r = rng.random_range(0..size.0);
51            let c = rng.random_range(0..size.1);
52            if !bomb_coords.contains(&(r, c)) {
53                bomb_coords.push((r, c));
54            }
55        }
56        bomb_coords
57    }
58
59    /// Create a board initialized with bombs at the given coordinates.
60    ///
61    /// `bomb_coords` is a slice of `(row, col)` tuples. Coordinates outside the board are ignored. After placing bombs the adjacent counts for number cells are computed.
62    pub fn from_bomb_coords(size: (usize, usize), bomb_coords: &[(usize, usize)]) -> Board {
63        let mut b = Board::new_game(size);
64
65        for &(r, c) in bomb_coords {
66            if let Some(idx) = b.rc_to_index_checked(r, c) {
67                b.board[idx] = Cell::new_bomb();
68            }
69        }
70
71        let (rows, cols) = size;
72        for r in 0..rows {
73            for c in 0..cols {
74                if let Some(idx) = b.rc_to_index_checked(r, c) {
75                    if b.board[idx].is_bomb() {
76                        continue;
77                    }
78                    let mut count = 0;
79                    for n_idx in b.neighbors_indices(idx) {
80                        if b.board[n_idx].is_bomb() {
81                            count += 1;
82                        }
83                    }
84                    b.board[idx] = Cell::new_number(count);
85                }
86            }
87        }
88
89        b
90    }
91
92    /// Convert (row, col) to linear index if in bounds.
93    fn rc_to_index_checked(&self, r: usize, c: usize) -> Option<usize> {
94        let (rows, cols) = self.size;
95        if r < rows && c < cols {
96            Some(r * cols + c)
97        } else {
98            None
99        }
100    }
101
102    /// Convert linear index to (row, col).
103    fn index_to_rc(&self, idx: usize) -> (usize, usize) {
104        let cols = self.size.1;
105        (idx / cols, idx % cols)
106    }
107
108    /// Return neighbor indices (8-way) for a given index.
109    fn neighbors_indices(&self, idx: usize) -> Vec<usize> {
110        let (r, c) = self.index_to_rc(idx);
111        let (rows, cols) = self.size;
112        let mut out = Vec::with_capacity(8);
113
114        let r_min = if r == 0 { 0 } else { r - 1 };
115        let c_min = if c == 0 { 0 } else { c - 1 };
116        let r_max = if r + 1 >= rows { rows - 1 } else { r + 1 };
117        let c_max = if c + 1 >= cols { cols - 1 } else { c + 1 };
118
119        for rr in r_min..=r_max {
120            for cc in c_min..=c_max {
121                if rr == r && cc == c {
122                    continue;
123                }
124                out.push(rr * cols + cc);
125            }
126        }
127
128        out
129    }
130
131    /// Count the number of bombs revealed around the current cell
132    pub fn count_flagged_neighbors(&mut self, idx: usize) -> usize {
133        self.neighbors_indices(idx)
134            .iter()
135            .filter(|&&n_idx| self.board[n_idx].is_flagged())
136            .count()
137    }

139    /// Reveal the cell at `idx`.
140    ///
141    /// Returns:
142    /// * `Ok(number)` if a number cell was revealed (the number value).
143    /// * `Err("BOMB")` if a bomb was revealed (and sets `game_over`).
144    /// * `Err("OUT_OF_BOUNDS")` for invalid indices.
145    ///
146    /// If the revealed cell is a zero (0) number, a flood fill reveal (BFS) is performed revealing contiguous zeroes and their border number cells.
147    pub fn reveal(&mut self, idx: usize) -> Result<i32, String> {
148        if idx >= self.board.len() {
149            return Err(String::from("OUT_OF_BOUNDS"));
150        }
151
152        if self.board[idx].is_revealed() {
153            if self.board[idx].number_value().unwrap_or(0)
154                == self.count_flagged_neighbors(idx) as i32
155            {
156                self.neighbors_indices(idx).iter().for_each(|&n_idx| {
157                    if !self.board[n_idx].is_revealed() {
158                        self.reveal(n_idx).unwrap_or_default();
159                    }
160                });
161            }
162            return match self.board[idx].number_value() {
163                Some(n) => Ok(n),
164                None => Err(String::from("BOMB")),
165            };
166        }
167
168        if self.board[idx].is_flagged() {
169            return Err(String::from("FLAGGED"));
170        }
171
172        match self.board[idx].reveal() {
173            CellType::Bomb => {
174                self.game_over = true;
175                Err(String::from("BOMB"))
176            }
177            CellType::Number(n) => {
178                if n == 0 {
179                    self.flood_reveal(idx);
180                }
181                Ok(n)
182            }
183        }
184    }
185
186    /// Reveal neighbors recursively starting from a zero cell using BFS.
187    ///
188    /// This reveals neighboring zeroes and stops at number > 0 cells (they are revealed).
189    fn flood_reveal(&mut self, start_idx: usize) {
190        let mut q = VecDeque::new();
191        let mut visited = vec![false; self.board.len()];
192        q.push_back(start_idx);
193        visited[start_idx] = true;
194
195        while let Some(idx) = q.pop_front() {
196            if self.board[idx].is_flagged() {
197                continue;
198            }
199
200            let ctype = self.board[idx].reveal();
201
202            match ctype {
203                CellType::Bomb => {
204                    continue;
205                }
206                CellType::Number(n) => {
207                    if n == 0 {
208                        for n_idx in self.neighbors_indices(idx) {
209                            if !visited[n_idx]
210                                && !self.board[n_idx].is_revealed()
211                                && !self.board[n_idx].is_flagged()
212                            {
213                                visited[n_idx] = true;
214                                q.push_back(n_idx);
215                            }
216                        }
217                    }
218                }
219            }
220        }
221    }

223    /// Cycle a flag state for the given cell index using the Hidden->Flag->Question->Hidden cycle.
224    ///
225    /// Returns
226    /// * `Ok(())` on success
227    /// * `Err("OUT_OF_BOUNDS")` if index invalid
228    /// * `Err("REVEALED")` if trying to change the status of an already revealed cell.
229    pub fn cycle_flag(&mut self, idx: usize) -> Result<(), String> {
230        if idx >= self.board.len() {
231            return Err(String::from("OUT_OF_BOUNDS"));
232        }
233
234        if self.board[idx].is_revealed() {
235            return Err(String::from("REVEALED"));
236        }
237
238        self.board[idx].cycle_status();
239        Ok(())
240    }

242    /// Check if the player has won: All non-bomb cells are revealed.
243    pub fn check_win(&self) -> bool {
244        for cell in &self.board {
245            if !cell.is_bomb() && !cell.is_revealed() {
246                return false;
247            }
248        }
249        true
250    }

252    /// Reveal by (row, col)
253    ///
254    /// This method is used because the board is stored as a 1-dimensional vector
255    pub fn reveal_rc(&mut self, r: usize, c: usize) -> Result<i32, String> {
256        if let Some(idx) = self.rc_to_index_checked(r, c) {
257            self.reveal(idx)
258        } else {
259            Err(String::from("OUT_OF_BOUNDS"))
260        }
261    }
262
263    /// Cycle flag by (row, col)
264    ///
265    /// This method is used because the board is stored as a 1-dimensional vector
266    pub fn cycle_flag_rc(&mut self, r: usize, c: usize) -> Result<(), String> {
267        if let Some(idx) = self.rc_to_index_checked(r, c) {
268            self.cycle_flag(idx)
269        } else {
270            Err(String::from("OUT_OF_BOUNDS"))
271        }
272    }
